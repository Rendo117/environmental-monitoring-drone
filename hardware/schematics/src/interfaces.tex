% ---------------- (a) MQ-2 ----------------
\node[font=\bfseries, anchor=west] at (-0.5,5.2) {(a) MQ-2 chemiresistor read-out};
\draw (0,4.2) node[left]{$V_C = 5$\,V} -- (3,4.2);
\draw (0.8,4.2) to[R, l_=$R_H\!\approx\!33\,\Omega$ (heater), *-] (0.8,0.4) node[ground]{};
\draw (3,4.2) to[vR, l=$R_S$ (SnO$_2$)] (3,2.2) coordinate (ao);
\draw (ao) to[R, l={$R_L = 1$\,k$\Omega$}] (3,0.4) node[ground]{};
\draw (ao) to[short, *-o] (5,2.2) node[right]{A3};

\node[align=left, font=\small, anchor=north west] at (-0.5,-0.2) {
$V_{out} = V_C\,\dfrac{R_L}{R_S+R_L}\;\Rightarrow\; R_S = R_L\,\dfrac{V_C - V_{out}}{V_{out}}$\\[6pt]
$V_{out} = \text{ADC}\cdot\dfrac{5\,\text{V}}{1023}$,\quad $R_0 = \dfrac{R_S^{\text{clean air}}}{9.83}$\\[6pt]
$\log_{10}\!\dfrac{R_S}{R_0} = m\,(\log_{10}c - x_0) + y_0$\\[4pt]
$c = 10^{\left(\log_{10}(R_S/R_0) - y_0\right)/m \,+\, x_0}$ [ppm]\\[4pt]
\footnotesize LPG: $(x_0,y_0,m)=(2.3,\,0.21,\,-0.47)$};

% ---------------- (b) battery ----------------

\node[align=left, font=\small, anchor=north west] at (-0.5,-0.2) {
$V_{A0} = (V_{bat}-V_D)\,\dfrac{R_b}{R_a+R_b} = 0.4\,(V_{bat}-0.8\,\text{V})$\\[6pt]
$V_{bat}\,[\text{cV}] = (\text{ADC} + 65)\cdot 1.2317$\\[6pt]
\footnotesize firmware low-pass: $v_k = 0.92\,v_{k-1} + 0.08\,(\text{ADC}+65)\cdot1.2317$\\
\footnotesize 12.6\,V full $\to$ ADC $\approx$ 966 (\,$<1023$, no clipping)};
